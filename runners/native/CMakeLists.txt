cmake_minimum_required(VERSION 3.20)
project(ckey_test_native CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(CKEY_SOURCE_DIR "" CACHE PATH "Pinned clearkeysolutions/ckey checkout (see configs/baseline.yaml)")
if(NOT EXISTS "${CKEY_SOURCE_DIR}/include/ckey/ckey.h")
  message(FATAL_ERROR "CKEY_SOURCE_DIR must point at a ckey checkout; got '${CKEY_SOURCE_DIR}'")
endif()
get_filename_component(CKT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

# Reproducibility: forbid FMA contraction so frames hash identically across CPUs.
add_compile_options(-ffp-contract=off)

option(CKT_SANITIZE "Build worker and ckey core with ASan + UBSan" OFF)
option(CKT_FUZZ "Build libFuzzer targets (clang only)" OFF)
if(CKT_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  add_link_options(-fsanitize=address,undefined)
endif()
if(CKT_FUZZ)
  add_compile_options(-fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
endif()

# The production core, built from the pinned checkout with its own CMake.
set(CKEY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CKEY_BUILD_TELEMETRY OFF CACHE BOOL "" FORCE)
add_subdirectory(${CKEY_SOURCE_DIR} ckey EXCLUDE_FROM_ALL)

execute_process(COMMAND git -C ${CKEY_SOURCE_DIR} rev-parse HEAD OUTPUT_VARIABLE CKT_CKEY_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git -C ${CKEY_SOURCE_DIR} status --porcelain OUTPUT_VARIABLE CKT_CKEY_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git -C ${CKT_REPO_ROOT} rev-parse HEAD OUTPUT_VARIABLE CKT_HARNESS_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git -C ${CKT_REPO_ROOT} status --porcelain -- simulation runners/native OUTPUT_VARIABLE CKT_HARNESS_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(CKT_CKEY_DIRTY)
  set(CKT_CKEY_SHA "${CKT_CKEY_SHA}-dirty")
endif()
if(NOT CKT_HARNESS_SHA)
  set(CKT_HARNESS_SHA "uncommitted")
endif()
if(CKT_HARNESS_DIRTY)
  set(CKT_HARNESS_SHA "${CKT_HARNESS_SHA}-dirty")
endif()
set(CKT_BUILD_INFO "{\"compiler\":\"${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\",\"build_type\":\"${CMAKE_BUILD_TYPE}\",\"sanitize\":\"${CKT_SANITIZE}\",\"system\":\"${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_PROCESSOR}\"}")
string(REPLACE "\"" "\\\"" CKT_BUILD_INFO_ESC "${CKT_BUILD_INFO}")

set(CKT_INCLUDES ${CKT_REPO_ROOT} ${CKEY_SOURCE_DIR}/include ${CKEY_SOURCE_DIR}/src)
set(CKT_DEFS CKT_CKEY_SHA="${CKT_CKEY_SHA}" CKT_HARNESS_SHA="${CKT_HARNESS_SHA}" CKT_BUILD_INFO="${CKT_BUILD_INFO_ESC}")

add_executable(ckt_worker worker.cpp selftest.cpp alloc_track.cpp)
target_include_directories(ckt_worker PRIVATE ${CKT_INCLUDES})
target_compile_definitions(ckt_worker PRIVATE ${CKT_DEFS})
target_link_libraries(ckt_worker PRIVATE ckey_scan ckey_render ckey_codec)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ckt_worker PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

if(CKT_FUZZ)
  foreach(t fuzz_codeword fuzz_gray8)
    add_executable(${t} ${CKT_REPO_ROOT}/security/fuzz/${t}.cpp)
    target_include_directories(${t} PRIVATE ${CKT_INCLUDES})
    target_link_libraries(${t} PRIVATE ckey_scan ckey_codec)
    target_link_options(${t} PRIVATE -fsanitize=fuzzer,address,undefined)
  endforeach()
endif()
